; Bool (core SMT)
;
#begin
(declare-const x Bool)
(declare-const y Bool)
(declare-const z Bool)
(assert (= z (xor x y)))
(assert (= x (xor z y)))
(assert (= y (xor z x)))
#end

; Finite field
;
#begin
(declare-const x F)
(declare-const y F)
(declare-const z F)
(assert (= z (ff.add x y)))
(assert (= x (ff.add z (ff.neg y))))
(assert (= y (ff.add z (ff.neg x))))
#end
